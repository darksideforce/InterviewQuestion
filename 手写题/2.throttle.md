```ts
function throttle(fn, delay) {
  let timer = null; // 闭包保存定时器状态（标志位）
  return function (...args) {
    // 如果定时器还在，说明还在“冷却期”，直接 return，什么都不做
    if (timer) return;
    
    // 如果没处于冷却期，开启定时器
    timer = setTimeout(() => {
      fn.apply(this, args);
      // 执行完毕后，清空定时器，表示“冷却完毕”，可以接受下一次触发
      timer = null; 
    }, delay);
  };
}
```
