```ts
function debounce(fn, delay) {
  let timer = null; // 闭包保存定时器
  // 返回一个被包装后的函数
  return function (...args) {
    // 每次触发时，如果还有定时器，说明在 delay 时间内又触发了，就把它清掉
    if (timer) {
      clearTimeout(timer);
    }
    
    // 重新开启一个新的定时器
    timer = setTimeout(() => {
      // 箭头函数没有自己的 this，这里的 this 指向外层闭包被调用时的执行上下文
      // 保证原函数 fn 能够正确接收到 this 和参数
      fn.apply(this, args);
    }, delay);
  };
}
```
